\begin{frame}
    \frametitle{Uranium Enrichment}
    \begin{itemize}
        \item Process to increase the relative abundance of specific
              isotopes of an element
        \item The US relies on gaseous centrifuge technology 
        \cite{us_nrc_centrus_2021,us_nrc_louisiana_2022}
        \item<2-> The throughput of a facility is based on the product 
              mass required and the \gls{SWU} capacity
            \begin{subequations}
                \begin{equation}
                    F = P + T
                \end{equation}
                \begin{equation}
                    x_fF = x_pP + x_tT
                    \label{eq:enrichment_assasys}
                \end{equation}
                \begin{equation}
                    SWU = \left[P*V(x_p) +T*V(x_t) - F*V(x_f)\right]*t
                    \label{eq:swu}
                \end{equation}
                \text{in which:}
                \begin{equation}
                    V(x_i) = (2x_i - 1)*\ln\left(\frac{x_i}{1-x_i}\right)
                    \label{eq:sep_potential}
                \end{equation}
                \label{eq:enrichment}
            \end{subequations}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Multiple efforts to estimate HALEU needs}
    \begin{itemize}
        \item Surveys from \gls{NEI} \cite{korsnick_updated_2021}
              estimate a cumulative need of 2,924.3 MTU by 2025
        \item Projects with a site selected will need an estimated 20 MTU 
              between 2024-2027 \cite{noauthor_establishing_2022}, and 
              potentially 78.7 MTU by 2027 \cite{korsnick_updated_2021}
        \item Fuel cycle models estimate needing 520 MTU/year of \gls{HALEU}
              to meet current net-zero carbon goals \cite{dixon_estimated_2022}
    \end{itemize}
\end{frame}

\subsection{Objectives}
\begin{frame}
    \frametitle{Objectives}
    \begin{itemize}
    \item This work explores how the deployment of \gls{HALEU}-fueled reactors 
    affects the US nuclear fuel cycle. 
    \item<2-> Model the transition from \glspl{LWR} to advanced reactors in 
          a once-through and recycling fuel cycle
    \begin{itemize}
        \item<2-> Mass of enriched uranium sent to reactors
        \item<2-> Mass of feed uranium required to produce enriched uranium
        \item<2-> \gls{SWU} capacity required to enrich uranium 
        \item<2-> Mass of \gls{SNF} for disposal
    \end{itemize}
    \item<3-> Perform sensitivity analysis and optimize select transition 
         scenarios
    \item<4-> Neutronics analysis of \gls{HALEU} created from impure \gls{HEU}
\end{itemize}
\end{frame}
